export default {
  async fetch(request, env) {
    // ---- CORS ----
    const corsHeaders = {
      "Access-Control-Allow-Origin": "https://santoshhdandin.github.io/dantaveda/",
      "Access-Control-Allow-Methods": "POST, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type",
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    if (request.method !== "POST") {
      return new Response("Only POST allowed", {
        status: 405,
        headers: corsHeaders,
      });
    }

    try {
      const body = await request.json();
      const userMessage = body.message;

      if (!userMessage) {
        return new Response(
          JSON.stringify({ reply: "Empty message" }),
          { headers: corsHeaders }
        );
      }

      const lyzrResponse = await fetch(
        "https://agent-prod.studio.lyzr.ai/v3/inference/chat/",
        {
          method: "POST",
          headers: {
            "Content-Type": "application/json",
            "x-api-key": env.LYZR_API_KEY,
          },
          body: JSON.stringify({
            user_id: "website-user",
            agent_id: env.LYZR_AGENT_ID,
            session_id: "website-session",
            message: userMessage,
          }),
        }
      );

      const data = await lyzrResponse.json();

      return new Response(
        JSON.stringify({
          reply: data.response || "No response from agent",
        }),
        { headers: corsHeaders }
      );

    } catch (err) {
      return new Response(
        JSON.stringify({ reply: "Agent error" }),
        { status: 500, headers: corsHeaders }
      );
    }
  },
};
